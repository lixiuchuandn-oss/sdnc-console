# 用户管理修改纪要（系统设置 / 权限管控 / 用户管理）
日期：2026-03-25

## 目标与范围
本次变更仅覆盖“系统用户（/api/users）”管理的删除/编辑行为，包含后端 API 约束与前端页面交互限制。

### 需求（最终确认口径）
1. 内置三权分立相关账号：禁止删除，但允许编辑；编辑时“姓名、用户名不可修改”（仅允许改邮箱/状态/密码）。
2. 自定义创建的系统用户：允许删除、允许编辑；编辑时“姓名、用户名不可修改”（仅允许改邮箱/状态/密码）。

## 修改后的效果（对外行为）
### 1) 删除规则
- `admin / sysadmin / auditor / secadmin`：删除直接被后端拒绝（返回 400），前端不提供删除入口。
- 其他系统用户：允许删除（前端可单删/批量删；后端正常执行删除）。

### 2) 编辑规则（系统用户 /api/users/:id）
- 所有系统用户：姓名（fullName）、用户名（username）一律不允许修改（后端强制校验，任何尝试改动都会返回 400）。
- 允许修改的字段（前端提交口径）：邮箱（email）、状态（status）、密码（password）。

## 影响面与注意事项
- 后端为强约束：即使绕过前端直接调用 API，只要试图修改 fullName/username 就会被拒绝。
- 前端编辑弹窗对所有“可编辑用户”默认进入“姓名/用户名只读模式”，仅展示可改项并提交。
- `admin` 账号在前端列表中仍不显示操作按钮（历史行为保留），同时后端也禁止删除内置账号（双保险）。

## 关键改动点（摘要）
### 后端（backend/src/routes/sysuser/index.ts）
1. 增加内置系统用户名集合：`admin / sysadmin / auditor / secadmin`，并在列表接口中返回 `builtIn` 标识。
2. 删除接口 `DELETE /api/users/:id`：若目标用户为内置账号，返回 `400 { message: '内置用户不允许删除' }`。
3. 更新接口 `PUT/PATCH /api/users/:id`：
   - 若请求体显式携带 `fullName` 且与原值不同：返回 `400 { message: '姓名不允许修改' }`
   - 若请求体显式携带 `username` 且与原值不同：返回 `400 { message: '用户名不允许修改' }`
   - 仍允许更新 `email/status/password` 等其他字段。

### 前端（frontend/src/pages/Access/Users.tsx）
1. 操作列：
   - `admin`：不显示操作按钮。
   - `builtIn=true`：仅提供“编辑”，不提供“删除用户”。
   - 其他：提供“编辑 + 删除用户”。
2. 编辑弹窗：
   - 所有通过“编辑”打开的用户均进入 `isEditingAdmin=true` 模式：姓名/用户名输入框禁用。
   - 保存时仅提交 `{ email, status, password }`，不提交 fullName/username，避免触发后端拒绝。

```ts
router.get('/me', async (req: AuthRequest, res: Response) => {
  try {
    if (!req.user) {
      return res.status(401).json({ message: '未授权' });
    }

    let resp: any = null;
    // Check if user is a tenant user (via isTenantUser flag from authMiddleware, or fallback to role check)
    if ((req.user as any).isTenantUser || (req.user as any).role === 'tenant') {
      // tenant user already populated in authMiddleware
      const tu = req.user as any;

      // Fetch full tenant/org info
      let tenantName = undefined;
      let organizationName = undefined;
      try {
        const tenantRepo = AppDataSource.getRepository(TenantUser);
        const fullUser = await tenantRepo.findOne({
          where: { id: tu.id },
          relations: ['tenant', 'organization'],
        });
        tenantName = fullUser?.tenant?.name;
        organizationName = fullUser?.organization?.name;
      } catch (e) {
        sysUserLog.warn('Failed to fetch tenant/org info', e instanceof Error ? e.message : String(e));
      }

      let permissions: string[] = [];
      try {
        const trId = tu.role;
        const assignRepo = AppDataSource.getRepository(RoleAssignment);
        const roleRepo = AppDataSource.getRepository(RBACRole);

        // 1. Get explicit assignments
        const assigns = await assignRepo.find({
          where: { userType: 'tenant', userId: tu.id },
        });
        const roleIds = assigns.map((a) => a.roleId);

        // 2. Include primary role if it's a UUID (custom role)
        if (trId && /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(trId)) {
          if (!roleIds.includes(trId)) {
            roleIds.push(trId);
          }
        }

        // 3. Fetch all roles and merge permissions
        if (roleIds.length > 0) {
          const roles = await roleRepo.find({ where: { id: In(roleIds) } });
          for (const r of roles) {
            if (r.permissions) {
              try {
                const raw = JSON.parse(r.permissions);
                if (Array.isArray(raw)) {
                  permissions = permissions.concat(raw);
                }
              } catch {}
            }
          }
        }

        // Deduplicate
        permissions = Array.from(new Set(permissions));
      } catch (e) {
        sysUserLog.warn('Failed to fetch permissions for tenant user', e instanceof Error ? e.message : String(e));
      }

      resp = {
        id: tu.id,
        username: tu.username,
        email: tu.email,
        fullName: tu.fullName,
        role: 'tenant',
        tenantRole: tu.role,
        permissions,
        status: tu.status,
        createdAt: tu.createdAt,
        lastLoginAt: tu.lastLoginAt,
        tenantName,
        organizationName,
        tenantId: tu.tenantId, // 租户ID
        organizationId: tu.organizationId, // 组织ID
      };
    } else {
      const userRepository = AppDataSource.getRepository(User);
      const user = await userRepository.findOne({
        where: { id: req.user.id },
        relations: ['accessibleClouds'],
      });
      if (!user) return res.status(404).json({ message: '用户不存在' });
      // 使用 authMiddleware 已经计算好的 permissions（包含 super_admin 的 ['*']）
      const permissions = Array.isArray((req.user as any).permissions) ? (req.user as any).permissions : [];
      resp = {
        id: user.id,
        username: user.username,
        email: user.email,
        fullName: user.fullName,
        role: user.role,
        status: user.status,
        accessibleClouds: user.accessibleClouds,
        createdAt: user.createdAt,
        lastLoginAt: user.lastLoginAt,
        permissions, // 返回权限列表
      };
    }

    res.json(resp);
  } catch (error) {
    res.status(500).json({ message: '获取用户信息失败' });
  }
});

router.post('/me/password', async (req: AuthRequest, res: Response) => {
  try {
    if (!req.user) {
      return res.status(401).json({ message: '未授权' });
    }

    if ((req.user as any).isTenantUser || (req.user as any).role === 'tenant') {
      return res.status(403).json({ message: '租户用户暂不支持此接口' });
    }

    const { oldPassword, newPassword } = req.body || {};
    const oldPwd = String(oldPassword || '').trim();
    const newPwd = String(newPassword || '').trim();

    if (!oldPwd || !newPwd) {
      return res.status(400).json({ message: '缺少 oldPassword 或 newPassword' });
    }
    if (newPwd.length < 8 || newPwd.length > 64) {
      return res.status(400).json({ message: '新密码长度需为 8-64 位' });
    }

    const userRepository = AppDataSource.getRepository(User);
    const user = await userRepository.findOne({ where: { id: req.user.id } });
    if (!user) {
      return res.status(404).json({ message: '用户不存在' });
    }

    const ok = await bcrypt.compare(oldPwd, user.password);
    if (!ok) {
      return res.status(400).json({ message: '原密码错误' });
    }

    user.password = await bcrypt.hash(newPwd, 10);
    await userRepository.save(user);

    res.json({ message: '密码修改成功' });
  } catch (error) {
    res.status(500).json({ message: '修改密码失败' });
  }
});

router.post('/', async (req: AuthRequest, res: Response) => {
  try {
    if (!req.user) return res.status(401).json({ message: '未授权' });
    const perms: string[] = Array.isArray((req.user as any).permissions) ? (req.user as any).permissions : [];
    if (!checkPermission(perms, 'users:write')) return res.status(403).json({ message: '权限不足' });
    const { username, email, password, fullName, role } = req.body;
    if (!username?.trim() || !email?.trim() || !password?.trim() || !fullName?.trim()) {
      return res.status(400).json({ message: '缺少必填字段：username/email/password/fullName' });
    }
    if (role && String(role) !== UserRole.ADMIN) {
      return res.status(400).json({ message: '创建系统用户时不允许指定为超级管理员，请先创建管理员后再按需赋权' });
    }
    const userRepository = AppDataSource.getRepository(User);
    const existed = await userRepository.findOne({ where: [{ username }, { email }] });
    if (existed) return res.status(400).json({ message: '用户名或邮箱已存在' });
    const hashed = await bcrypt.hash(password, 10);
    const user = userRepository.create({ username, email, password: hashed, fullName, role: UserRole.ADMIN });
    const saved = await userRepository.save(user);
    res.status(201).json({
      id: saved.id,
      username: saved.username,
      email: saved.email,
      fullName: saved.fullName,
      role: saved.role,
    });
  } catch (e: any) {
    res.status(500).json({ message: '创建用户失败', error: String(e) });
  }
});

// 超级管理员：删除系统用户
router.delete('/:id', async (req: AuthRequest, res: Response) => {
  try {
    if (!req.user) return res.status(401).json({ message: '未授权' });
    const perms: string[] = Array.isArray((req.user as any).permissions) ? (req.user as any).permissions : [];
    if (!checkPermission(perms, 'users:write')) return res.status(403).json({ message: '权限不足' });
    const { id } = req.params;
    const userRepository = AppDataSource.getRepository(User);
    const user = await userRepository.findOne({ where: { id } });
    if (!user) {
      return res.status(404).json({ message: '用户不存在' });
    }
    if (req.user.id === user.id) {
      return res.status(400).json({ message: '不能删除当前登录的账户' });
    }
    if (isBuiltInSystemUser(user.username)) {
      return res.status(400).json({ message: '内置用户不允许删除' });
    }
    if (user.role === UserRole.SUPER_ADMIN) {
      const count = await userRepository.count({ where: { role: UserRole.SUPER_ADMIN } });
      if (count <= 1) {
        return res.status(400).json({ message: '不能删除最后一个超级管理员' });
      }
    }
    const assignRepo = AppDataSource.getRepository(RoleAssignment);
    await assignRepo.delete({ userType: 'system', userId: id });
    await userRepository.remove(user);
    res.json({ message: '用户删除成功' });
  } catch (error) {
    res.status(500).json({ message: '删除用户失败' });
  }
});

// 编辑系统用户的处理函数（PUT 和 PATCH 共用）
const updateUserHandler = async (req: AuthRequest, res: Response) => {
  try {
    if (!req.user) return res.status(401).json({ message: '未授权' });
    const perms: string[] = Array.isArray((req.user as any).permissions) ? (req.user as any).permissions : [];
    if (!checkPermission(perms, 'users:write')) return res.status(403).json({ message: '权限不足' });
    const { id } = req.params;
    const { username, email, fullName, role, status, password } = req.body as any;
    const userRepository = AppDataSource.getRepository(User);
    const user = await userRepository.findOne({ where: { id } });
    if (!user) return res.status(404).json({ message: '用户不存在' });

    if (Object.prototype.hasOwnProperty.call(req.body || {}, 'fullName') && fullName !== user.fullName) {
      return res.status(400).json({ message: '姓名不允许修改' });
    }

    if (Object.prototype.hasOwnProperty.call(req.body || {}, 'username') && username !== user.username) {
      return res.status(400).json({ message: '用户名不允许修改' });
    }

    if (email && email !== user.email) {
      const existed = await userRepository.findOne({ where: { email } });
      if (existed) return res.status(400).json({ message: '邮箱已存在' });
      user.email = email;
    }
    if (role) user.role = role;
    if (status) user.status = status;
    if (password && String(password).trim()) {
      user.password = await bcrypt.hash(String(password).trim(), 10);
    }

    const saved = await userRepository.save(user);
    res.json({
      id: saved.id,
      username: saved.username,
      email: saved.email,
      fullName: saved.fullName,
      role: saved.role,
      status: saved.status,
      password: saved.password,
    });
  } catch (e: any) {
    res.status(500).json({ message: '更新用户失败', error: String(e) });
  }
};

// 超级管理员：编辑系统用户（支持更新基础信息与密码）
router.put('/:id', updateUserHandler);
router.patch('/:id', updateUserHandler);
```

```tsx
                  fetchUsers(true); // 强制刷新，跳过缓存
                } catch (error: any) {
                  message.error(error?.message || '删除失败');
                }
              } },
          ];
      // Filter out invalid items or dividers if needed
      const items = itemsRaw.filter(Boolean);
      return <ResourceActionsDropdown items={items as any} />;
    } }
  ], [rolesByUser, editForm]);

  // 使用新的 hook 管理列设置
  const { displayColumns: visibleColumns } = useTableColumnSettings(defaultColumns, 'sys_users_v1');

  const handleBatchDelete = () => {
    if (selectedRowKeys.length === 0) return;
    Modal.confirm({
      title: '批量删除',
      content: `确定要删除选中的 ${selectedRowKeys.length} 个用户吗？`,
      okButtonProps: { danger: true },
      onOk: async () => {
        try {
          setLoading(true);
          await Promise.all(selectedRowKeys.map(id => 
            apiDelete(`/api/users/${id}`)
          ));
          message.success('批量删除成功');
          setSelectedRowKeys([]);
          fetchUsers(true); // 强制刷新，跳过缓存
        } catch (e) {
          message.error('部分或全部删除失败');
        } finally {
          setLoading(false);
        }
      }
    });
  };

  const batchMenuProps = {
    items: [
      { key: 'delete', label: '批量删除', danger: true, onClick: handleBatchDelete },
    ]
  };

  const fetchUsers = async (skipCache: boolean = false) => {
    try {
      setLoading(true);
      const json = await apiRequest('/api/users', skipCache ? { skipCache: true } : undefined);
      setUsers(Array.isArray(json) ? json : Array.isArray(json?.users) ? json.users : []);
    } catch {
      message.error('获取系统用户失败');
    } finally { setLoading(false); }
  };

  useEffect(() => { fetchUsers(); }, []);

  useEffect(() => {
    const run = async () => {
      if (!users.length) return;
      try {
        // 获取所有角色用于显示
        const rolesJson = await apiRequest('/api/system/rbac/roles').catch(() => []);
        const roles = Array.isArray(rolesJson) ? rolesJson : (Array.isArray(rolesJson?.roles) ? rolesJson.roles : []);

        const jsons = await Promise.all(users.map(u => 
          apiRequest(`/api/system/rbac/assignments?userType=system&userId=${encodeURIComponent(u.id)}`).catch(() => ({ assignments: [] }))
        ));
        const next: Record<string, RBACRole[]> = {};
        users.forEach((u, i) => {
          const list: Assignment[] = Array.isArray(jsons[i]?.assignments) ? jsons[i].assignments : [];
          const ids = list.map(it => it.roleId);
          const roleObjs = ids.map(id => roles.find((r: RBACRole) => r.id === id)).filter(Boolean) as RBACRole[];
          next[u.id] = roleObjs;
        });
        setRolesByUser(next);
      } catch {}
    };
    run();
  }, [users]);

  const filteredUsers = useMemo(() => {
    const lower = kw.toLowerCase();
    return users
      .filter(u => statusFilter === 'all' ? true : (String(u.status) === statusFilter))
      .filter(u => (u.username||'').toLowerCase().includes(lower) || (u.fullName||'').toLowerCase().includes(lower) || (u.email||'').toLowerCase().includes(lower));
  }, [users, kw, statusFilter]);

  const handleCreate = React.useCallback(async (values: any) => {
    try {
      await apiPost('/api/users', {
        username: values.username,
        email: values.email,
        password: values.password,
        fullName: values.fullName,
      });
      message.success('系统用户创建成功');
      setCreateOpen(false);
      createForm.resetFields();
      fetchUsers(true); // 强制刷新，跳过缓存
    } catch (error: any) {
      message.error(error?.message || '创建失败');
    }
  }, [createForm]);

  return (
    <PageContainer
      titleLevel={3}
      asCard={false}
    >
      <div className="page-toolbar">
        <Form 
          form={filterForm}
          className="toolbar-container" 
          layout="vertical" 
          onValuesChange={(_,all)=> { setKw(all.keyword||''); setStatusFilter(all.status||'active'); }}
        >
          <Form.Item name="keyword"><Input allowClear placeholder="搜索用户名/姓名/邮箱" className="toolbar-search" suffix={<SearchOutlined style={{ color: 'rgba(0,0,0,.45)' }} />} /></Form.Item>
          <Form.Item name="status" initialValue={statusFilter}><Select allowClear placeholder="全部" className="toolbar-filter-sm" options={[{ value:'all', label:'全部' }, { value:'active', label:'活跃' }, { value:'inactive', label:'非活跃' }, { value:'suspended', label:'暂停' }]} /></Form.Item>
          <Button type="link" className="toolbar-reset-btn" onClick={() => {
            setKw('');
            setStatusFilter('active');
            filterForm.resetFields();
            filterForm.setFieldsValue({ status: 'active' });
          }}>重置</Button>
        </Form>
        <div className="toolbar-actions">
          <Button type="primary" onClick={() => { const rand = `user-${Date.now().toString(36)}-${Math.random().toString(36).slice(2,6)}`; createForm.setFieldsValue({ username: rand }); setCreateOpen(true); }}>新建系统用户</Button>
          <Dropdown menu={batchMenuProps}>
            <Button icon={<EllipsisOutlined />} />
          </Dropdown>
          <TableColumnSettingsButton allColumns={defaultColumns} settingsKey="sys_users_v1" />
          <Button icon={<ReloadOutlined />} onClick={() => fetchUsers(true)} />
        </div>
      </div>

      <Table
        className="std-table"
        rowKey="id"
        loading={loading}
        dataSource={filteredUsers}
        columns={visibleColumns}
        scroll={{ 
          x: 'max-content',
        }}
        rowSelection={{
          selectedRowKeys,
          onChange: setSelectedRowKeys,
          getCheckboxProps: (record: SysUser) => ({
            disabled: (typeof record.builtIn === 'boolean') ? record.builtIn : (record.username === 'admin'),
          }),
        }}
        pagination={{
          size: 'small',
          showTotal: (total) => `共 ${total} 条`,
        }}
        bordered={false}
      />

      <StdFormModal title="新建系统用户" open={createOpen} onCancel={()=> setCreateOpen(false)} okText="创建" width={600} form={createForm} onFinish={handleCreate}>
        <Form.Item name="fullName" label="姓名" rules={[{ required: true, message: '请输入姓名' }]}> 
          <Input placeholder="例如：运维管理员" />
        </Form.Item>
        <Form.Item name="username" label="用户名" rules={[{ required: true, message: '请输入用户名' }]}> 
          <Input placeholder="例如：ops_admin" />
        </Form.Item>
        <Form.Item name="email" label="邮箱" rules={[{ required: true, message: '请输入邮箱' }, { type:'email', message: '请输入有效邮箱' }]}> 
          <Input placeholder="例如：ops@example.com" />
        </Form.Item>
        <Form.Item name="password" label="密码" rules={[{ required: true, message: '请输入密码' }]}> 
          <Input.Password placeholder="至少 8 位密码" />
        </Form.Item>
      </StdFormModal>

      <StdFormModal title="编辑用户" open={editOpen} onCancel={()=> { setEditOpen(false); setEditingUser(null); setIsEditingAdmin(false); }} okText="保存" width={600} form={editForm} onFinish={async (vals)=> {
        if (!editingUser) return;
        try {
          const body = isEditingAdmin
            ? { email: vals.email, status: vals.status, password: vals.password }
            : { fullName: vals.fullName, username: vals.username, email: vals.email, status: vals.status, password: vals.password };
          await apiPatch(`/api/users/${editingUser.id}`, body);
          message.success('用户更新成功');
          setEditOpen(false);
          setEditingUser(null);
          setIsEditingAdmin(false);
          fetchUsers(true); // 强制刷新，跳过缓存
        } catch (error: any) {
          message.error(error?.message || '更新失败');
        }
      }}>
        {isEditingAdmin ? (
          <>
            <Form.Item name="fullName" label="姓名">
              <Input disabled />
            </Form.Item>
            <Form.Item name="username" label="用户名">
              <Input disabled />
            </Form.Item>
            <Form.Item name="status" label="状态">
              <Select options={[{ value:'active', label:'活跃' }, { value:'inactive', label:'非活跃' }, { value:'suspended', label:'暂停' }]} />
            </Form.Item>
            <Form.Item name="email" label="邮箱" rules={[{ type:'email', message: '请输入有效邮箱' }]}> 
              <Input />
            </Form.Item>
            <Form.Item name="password" label="重置密码（留空则不修改）">
              <Input.Password placeholder="可选，输入则更新密码" />
            </Form.Item>
          </>
        ) : (
          <>
            <Form.Item name="fullName" label="姓名" rules={[{ required: true, message: '请输入姓名' }]}> 
              <Input />
            </Form.Item>
            <Form.Item name="username" label="用户名" rules={[{ required: true, message: '请输入用户名' }]}> 
              <Input />
            </Form.Item>
            <Form.Item name="email" label="邮箱" rules={[{ type:'email', message: '请输入有效邮箱' }]}> 
              <Input />
            </Form.Item>
            <Form.Item name="status" label="状态">
              <Select options={[{ value:'active', label:'活跃' }, { value:'inactive', label:'非活跃' }, { value:'suspended', label:'暂停' }]} />
            </Form.Item>
            <Form.Item name="password" label="重置密码（留空则不修改）">
              <Input.Password placeholder="可选，输入则更新密码" />
            </Form.Item>
          </>
        )}
      </StdFormModal>
    </PageContainer>
  );
};

export default AccessUsersPage;
```
