# 资源总览-虚拟机统计（“已关机/统计中”）过滤回收站虚拟机 - 修改纪要

- 需求：资源总览中的虚拟机统计列表中，“已关机/统计中”需过滤掉回收站中的虚拟机
- 结论：在统计逻辑中排除带有回收站标识的虚拟机（finalizers 或 labels 标记为 tower.io/recycle-bin）

## 修改点概述
- 新增回收站识别函数 isRecycleBinVm，统一判断 VM 是否在回收站
- 在虚拟机统计聚合入口 parseVMInstancesToStatusCounts 中，对 VM 列表先行过滤回收站 VM
- 补充单元测试，覆盖回收站 VM 的过滤行为

## 修改后的效果
- 总数 total 不再计入回收站中的虚拟机
- 分段统计（运行中/已暂停/已关机/其他）均不包含回收站中的虚拟机
- 资源总览的“虚拟机”卡片显示值随之正确更新（依赖上述统计函数）

export interface VMInstancesSummary {
  total: number;
  segments: VMStatusSegment[];
}

function isRecycleBinVm(vm: any): boolean {
  const finalizers = vm?.metadata?.finalizers || [];
  const labels = vm?.metadata?.labels || {};
  return finalizers.includes('tower.io/recycle-bin') || labels['tower.io/recycle-bin'] === 'true';
}

export function parseVMInstancesToStatusCounts(vmResponse: any, vmiResponse?: any): VMInstancesSummary {
  const vmList = getVMList(vmResponse).filter((vm) => !isRecycleBinVm(vm));
  const vmiList = getVMIList(vmiResponse);
  // ...后续统计逻辑保持不变
  return { total: vmList.length, segments };
}
```

### 3) 单元测试 vmInstances.test.ts（新增）

// 文件：frontend/src/pages/Clouds/Harvester/Monitoring/utils/vmInstances.test.ts
import { parseVMInstancesToStatusCounts } from './vmInstances';

function seg(summary: any): Record<string, number> {
  return (summary?.segments || []).reduce((acc: any, s: any) => {
    acc[s.key] = s.value;
    return acc;
  }, {});
}

describe('parseVMInstancesToStatusCounts', () => {
  it('filters recycle-bin VMs out of total and segments', () => {
    const vmResponse = {
      items: [
        { metadata: { namespace: 'ns1', name: 'vm-running' } },
        { metadata: { namespace: 'ns1', name: 'vm-stopped' }, status: { printableStatus: 'Stopped' } },
        {
          metadata: {
            namespace: 'ns1',
            name: 'vm-recycled',
            finalizers: ['tower.io/recycle-bin'],
          },
          status: { printableStatus: 'Stopped' },
        },
      ],
    };

    const vmiResponse = {
      items: [
        { metadata: { namespace: 'ns1', name: 'vm-running' }, status: { phase: 'Running', conditions: [] } },
      ],
    };

    const summary = parseVMInstancesToStatusCounts(vmResponse, vmiResponse);
    const m = seg(summary);

    expect(summary.total).toBe(2);
    expect(m.running).toBe(1);
    expect(m.stopped).toBe(1);
  });
});
```
