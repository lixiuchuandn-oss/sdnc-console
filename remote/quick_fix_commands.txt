# 快速修复数据库问题的命令（在服务器上执行）

# 1. 停止应用
pm2 stop compute-tower || true
pm2 delete compute-tower || true

# 2. 启用数据库同步并重启应用
cd /root/compute-tower
DB_SYNC=true DB_PATH=/root/compute-tower/data.sqlite NODE_ENV=production pm2 start dist/src/index.js --name compute-tower

# 3. 保存 PM2 配置
pm2 save

# 4. 查看应用状态
pm2 status

# 5. 查看应用日志（确认数据库初始化成功）
pm2 logs compute-tower --lines 50
